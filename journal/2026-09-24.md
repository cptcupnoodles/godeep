Appending subsequent structures alters the slice header length parameter without triggering systemic memory reallocation arrays across the heap.
- **Slice Header Geometry:** Creating the target sub-slice using boundary segmentation (`registry[2:]`) yields an immediate length of `2` but maps a persistent capacity structural value of `3`. This proves that capacity tracking maps linearly from the designated offset start index to the absolute terminal pointer boundary of the underlying allocation block.
- **Synchronized Address References:** Modifying elements through `serverPtr := &registry[2]` operates entirely via direct memory reference updates. Because the window slice `highPriorityTable[0]` evaluates the identical hardware storage coordinates, structural changes update automatically across all corresponding views. This is verified by the absolute matching hex registers (`0xc00012e060`).

import "fmt"

type Server struct {
	IP        string
	Weight    int
	IsHealthy bool
}

	registry := make([]Server, 0, 5)
	registry = append(registry,
		Server{IP: "192.168.1.10", Weight: 10, IsHealthy: true},
		Server{IP: "192.168.1.11", Weight: 20, IsHealthy: true},
		Server{IP: "192.168.1.12", Weight: 30, IsHealthy: false},
		Server{IP: "192.168.1.13", Weight: 40, IsHealthy: true},
	)

	highPriorityTable := registry[2:]
	serverPtr := &registry[2]
	serverPtr.IsHealthy, serverPtr.Weight = true, 50

	fmt.Printf("Registry: %p vs highPT: %p \n", &registry[2], &highPriorityTable[0])
}
